### DB-04: Android AWG socket protection accepts zero usable sockets

Severity: **Medium**  
Affected target: Android and Android TV AmneziaWG path  
State: **Verified static defect**

Evidence: `android/app/src/main/kotlin/org/winsvpn/app/AwgSocketProtection.kt:27-48` skips every descriptor below zero and returns success after the loop. `VpnTunnelService.kt:3450-3462` requests IPv4 and IPv6 descriptors, then publishes the engine if this helper returns. Upstream AmneziaWG returns `-1` when the handle is invalid, the bind object cannot expose descriptors, or descriptor lookup fails. Root cause: unavailable address families are treated individually, but the helper has no aggregate invariant requiring at least one usable, protected carrier socket.

Replace `protectOrRollback` with this complete implementation:

```kotlin
@Throws(IOException::class)
fun protectOrRollback(
    sockets: () -> Iterable<AwgSocket>,
    protect: (Int) -> Boolean,
    rollback: () -> Unit,
    onUnavailable: (String) -> Unit = {},
) {
    try {
        var availableSocketCount = 0
        var protectedSocketCount = 0
        for (socket in sockets()) {
            if (socket.fd < 0) {
                onUnavailable(socket.label)
                continue
            }

            availableSocketCount++
            val protected = try {
                protect(socket.fd)
            } catch (failure: Exception) {
                throw IOException(
                    "VpnService.protect() failed for the AmneziaWG ${socket.label} socket",
                    failure,
                )
            }
            if (!protected) {
                throw IOException(
                    "VpnService.protect() refused the AmneziaWG ${socket.label} socket; " +
                        "refusing to keep an unprotected engine running.",
                )
            }
            protectedSocketCount++
        }
        if (availableSocketCount == 0 || protectedSocketCount != availableSocketCount) {
            throw IOException(
                "AmneziaWG exposed no carrier socket that could be protected; " +
                    "refusing to keep the engine running.",
            )
        }
    } catch (failure: Exception) {
        try {
            rollback()
        } catch (rollbackFailure: Exception) {
            failure.addSuppressed(rollbackFailure)
        }
        throw failure
    }
}
```

```python
def normalize_delivery_email(raw: str, *, required: bool) -> str:
    value = (raw or "").strip().lower()
    if not value:
        if required:
            raise ValueError("delivery email is required")
        return ""
    if len(value) > 254:
        raise ValueError("delivery email is too long")
    if any(ord(char) < 32 or ord(char) == 127 for char in value):
        raise ValueError("delivery email contains control characters")
    if not DELIVERY_EMAIL_RE.fullmatch(value):
        raise ValueError("invalid delivery email")
    local_part, _ = value.rsplit("@", 1)
    if len(local_part) > 64:
        raise ValueError("delivery email local part is too long")
    return value

def normalize_monero_buyer_email(raw: str) -> str:
    return normalize_delivery_email(raw, required=True)
```

### DB-06: Cloudflare injected JavaScript conflicts with CSP

Severity: **Medium**  
Affected target: live website CSP and Cloudflare JavaScript Detections

```javascript
function withSecurityHeaders(response, request) {
  const headers = new Headers(response.headers);
  const isHttps = new URL(request.url).protocol === "https:";
  if (isHttps) {
    headers.set("Strict-Transport-Security", "max-age=63072000; includeSubDomains");
  } else {
    headers.delete("Strict-Transport-Security");
  }
  if (!headers.has("X-Content-Type-Options")) headers.set("X-Content-Type-Options", "nosniff");
  if (!headers.has("X-Frame-Options")) headers.set("X-Frame-Options", "DENY");
  if (!headers.has("Referrer-Policy")) {
    headers.set("Referrer-Policy", "strict-origin-when-cross-origin");
  }
  if (!headers.has("Permissions-Policy")) {
    headers.set(
      "Permissions-Policy",
      "accelerometer=(), camera=(), geolocation=(), gyroscope=(), magnetometer=(), microphone=(), payment=(), usb=()"
    );
  }
  if (!headers.has("Cross-Origin-Opener-Policy")) {
    headers.set("Cross-Origin-Opener-Policy", "same-origin");
  }
  if (!headers.has("Cross-Origin-Resource-Policy")) {
    headers.set("Cross-Origin-Resource-Policy", "same-origin");
  }
  if (!headers.has("Origin-Agent-Cluster")) {
    headers.set("Origin-Agent-Cluster", "?1");
  }
  if (!headers.has("X-Permitted-Cross-Domain-Policies")) {
    headers.set("X-Permitted-Cross-Domain-Policies", "none");
  }
  const contentType = headers.get("Content-Type") || "";
  if (contentType.includes("text/html")) {
    if (!headers.has("Content-Security-Policy")) {
      headers.set("Content-Security-Policy", CONTENT_SECURITY_POLICY);
    }
    const directives = (headers.get("Cache-Control") || "")
      .split(",")
      .map((value) => value.trim())
      .filter(Boolean);
    if (!directives.some((value) => value.toLowerCase() === "no-transform")) {
      directives.push("no-transform");
    }
    headers.set("Cache-Control", directives.join(", "));
  }
  return new Response(response.body, {
    status: response.status,
    statusText: response.statusText,
    headers,
  });
}
```
